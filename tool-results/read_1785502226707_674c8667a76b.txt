   142→// USERS & RESUMES SECTION
   143→// ============================================================
   144→export function UsersSection({ token, onRefresh }: { token: string; onRefresh: () => void }) {
   145→  const [users, setUsers] = useState<AdminUser[]>([]);
   146→  const [loading, setLoading] = useState(true);
   147→  const [search, setSearch] = useState("");
   148→  const [planFilter, setPlanFilter] = useState<string>("");
   149→  const [roleFilter, setRoleFilter] = useState<string>("");
   150→  const [selectedUser, setSelectedUser] = useState<AdminUser | null>(null);
   151→  const [userDetailOpen, setUserDetailOpen] = useState(false);
   152→  const [editOpen, setEditOpen] = useState(false);
   153→  const [deleteOpen, setDeleteOpen] = useState(false);
   154→
   155→  const loadUsers = useCallback(async () => {
   156→    setLoading(true);
   157→    try {
   158→      const params = new URLSearchParams();
   159→      if (search) params.set("q", search);
   160→      if (planFilter && planFilter !== "all") params.set("plan", planFilter);
   161→      if (roleFilter && roleFilter !== "all") params.set("role", roleFilter);
   162→      const res = await fetch(`/api/admin/users?${params}`, { headers: authHeaders(token) });
   163→      if (!res.ok) throw new Error("Failed");
   164→      const json = await res.json();
   165→      setUsers(json.users);
   166→    } catch {
   167→      toast.error("Failed to load users");
   168→    } finally {
   169→      setLoading(false);
   170→    }
   171→  }, [token, search, planFilter, roleFilter]);
   172→
   173→  useEffect(() => { loadUsers(); }, [loadUsers]);
   174→
   175→  const openUserDetail = async (u: AdminUser) => {
   176→    setSelectedUser(u);
   177→    setUserDetailOpen(true);
   178→  };
   179→
   180→  const handleDelete = async () => {
   181→    if (!selectedUser) return;
   182→    try {
   183→      const res = await fetch(`/api/admin/users/${selectedUser.id}`, { method: "DELETE", headers: authHeaders(token) });
   184→      if (!res.ok) throw new Error("Failed");
   185→      toast.success("User deleted");
   186→      setDeleteOpen(false);
   187→      setUserDetailOpen(false);
   188→      loadUsers();
   189→      onRefresh();
   190→    } catch {
   191→      toast.error("Failed to delete user");
   192→    }
   193→  };
   194→
   195→  return (
   196→    <div className="space-y-4">
   197→      {/* Filters */}
   198→      <Card className="p-4 bg-slate-900 border-slate-800">
   199→        <div className="flex flex-wrap items-center gap-3">
   200→          <div className="relative flex-1 min-w-[200px]">
   201→            <Search className="w-4 h-4 absolute left-3 top-1/2 -translate-y-1/2 text-slate-500" />
   202→            <Input
   203→              value={search}
   204→              onChange={(e) => setSearch(e.target.value)}
   205→              placeholder="Search by email, name, or student ID..."
   206→              className="pl-9 bg-slate-800 border-slate-700 text-white h-9"
   207→              onKeyDown={(e) => e.key === "Enter" && loadUsers()}
   208→            />
   209→          </div>
   210→          <Select value={planFilter || "all"} onValueChange={(v) => setPlanFilter(v === "all" ? "" : v)}>
   211→            <SelectTrigger className="w-[140px] bg-slate-800 border-slate-700 text-white h-9"><SelectValue placeholder="All plans" /></SelectTrigger>
   212→            <SelectContent className="bg-slate-900 border-slate-700">
   213→              <SelectItem value="all">All plans</SelectItem>
   214→              {PLAN_OPTIONS.map((p) => <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>)}
   215→            </SelectContent>
   216→          </Select>
   217→          <Select value={roleFilter || "all"} onValueChange={(v) => setRoleFilter(v === "all" ? "" : v)}>
   218→            <SelectTrigger className="w-[140px] bg-slate-800 border-slate-700 text-white h-9"><SelectValue placeholder="All roles" /></SelectTrigger>
   219→            <SelectContent className="bg-slate-900 border-slate-700">
   220→              <SelectItem value="all">All roles</SelectItem>
   221→              <SelectItem value="user">User</SelectItem>
   222→              <SelectItem value="student">Student</SelectItem>
   223→              <SelectItem value="org_admin">Org Admin</SelectItem>
   224→            </SelectContent>
   225→          </Select>
   226→          <Button onClick={loadUsers} variant="outline" size="sm" className="bg-slate-800 border-slate-700 text-slate-300 h-9 gap-1.5">
   227→            <Search className="w-3.5 h-3.5" /> Search
   228→          </Button>
   229→          <div className="text-xs text-slate-400 ml-auto">{users.length} users</div>
   230→        </div>
   231→      </Card>
   232→
   233→      {/* Users list */}
   234→      <Card className="bg-slate-900 border-slate-800">
   235→        {loading ? (
   236→          <div className="flex items-center justify-center py-12"><Loader2 className="w-5 h-5 animate-spin text-teal-400" /></div>
   237→        ) : users.length === 0 ? (
   238→          <p className="text-center text-sm text-slate-400 py-12">No users found.</p>
   239→        ) : (
   240→          <div className="divide-y divide-slate-800">
   241→            {users.map((u) => (
   242→              <div key={u.id} className="flex items-center gap-3 p-4 hover:bg-slate-800/40 transition-colors">
   243→                <div className="w-9 h-9 rounded-full bg-gradient-to-br from-slate-700 to-slate-800 flex items-center justify-center text-sm font-semibold shrink-0">
   244→                  {u.email[0]?.toUpperCase()}
   245→                </div>
   246→                <div className="flex-1 min-w-0">
   247→                  <div className="flex items-center gap-2">
   248→                    <p className="text-sm font-medium truncate">{u.email}</p>
   249→                    {u.role === "student" && <Badge className="bg-violet-900/50 text-violet-300 border-0 text-[9px]"><GraduationCap className="w-2.5 h-2.5 mr-0.5" />Student</Badge>}
   250→                    {u.organization && <Badge className="bg-slate-800 text-slate-300 border-0 text-[9px]">{u.organization.name}</Badge>}
   251→                  </div>
   252→                  <p className="text-[11px] text-slate-400 mt-0.5">
   253→                    {u.name || "—"} · {u.resumeCount} resumes · {u.studentId ? `ID: ${u.studentId} · ` : ""}Joined {new Date(u.createdAt).toLocaleDateString()}
   254→                  </p>
   255→                </div>
   256→                <Badge className={`border-0 text-[10px] ${planBadge(u.plan)}`}>{planLabel(u.plan)}</Badge>
   257→                {u.planExpiresAt && new Date(u.planExpiresAt) < new Date() && (
   258→                  <Badge className="bg-rose-900/50 text-rose-300 border-0 text-[10px]">Expired</Badge>
   259→                )}
   260→                <DropdownMenu>
   261→                  <DropdownMenuTrigger asChild>
   262→                    <Button variant="ghost" size="sm" className="h-8 w-8 p-0 text-slate-400 hover:bg-slate-800">
   263→                      <Pencil className="w-3.5 h-3.5" />
   264→                    </Button>
   265→                  </DropdownMenuTrigger>
   266→                  <DropdownMenuContent align="end" className="bg-slate-900 border-slate-700">
   267→                    <DropdownMenuLabel className="text-slate-400 text-xs">Actions</DropdownMenuLabel>
   268→                    <DropdownMenuItem onClick={() => openUserDetail(u)} className="text-slate-200 text-xs cursor-pointer">
   269→                      <Eye className="w-3.5 h-3.5 mr-2" /> View Details & Resumes
   270→                    </DropdownMenuItem>
   271→                    <DropdownMenuItem onClick={() => { setSelectedUser(u); setEditOpen(true); }} className="text-slate-200 text-xs cursor-pointer">
   272→                      <Pencil className="w-3.5 h-3.5 mr-2" /> Edit Account
   273→                    </DropdownMenuItem>
   274→                    <DropdownMenuSeparator className="bg-slate-800" />
   275→                    <DropdownMenuItem onClick={() => { setSelectedUser(u); setDeleteOpen(true); }} className="text-rose-400 text-xs cursor-pointer">
   276→                      <Trash2 className="w-3.5 h-3.5 mr-2" /> Delete Account
   277→                    </DropdownMenuItem>
   278→                  </DropdownMenuContent>
   279→                </DropdownMenu>
   280→              </div>
   281→            ))}
   282→          </div>
   283→        )}
   284→      </Card>
   285→
   286→      {/* User detail dialog */}
   287→      {selectedUser && (
   288→        <UserDetailDialog
   289→          key={selectedUser.id}
   290→          user={selectedUser}
   291→          token={token}
   292→          open={userDetailOpen}
   293→          onOpenChange={setUserDetailOpen}
   294→          onEdit={() => { setUserDetailOpen(false); setEditOpen(true); }}
   295→          onDelete={() => { setUserDetailOpen(false); setDeleteOpen(true); }}
   296→        />
   297→      )}
   298→
   299→      {/* Edit dialog */}
   300→      {selectedUser && (
   301→        <EditUserDialog
   302→          user={selectedUser}
   303→          token={token}
   304→          open={editOpen}
   305→          onOpenChange={setEditOpen}
   306→          onSaved={() => { setEditOpen(false); loadUsers(); onRefresh(); }}
   307→        />
   308→      )}
   309→
   310→      {/* Delete confirm */}
   311→      <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   312→        <AlertDialogContent className="bg-slate-900 border-slate-700">
   313→          <AlertDialogHeader>
   314→            <AlertDialogTitle className="text-white">Delete user account?</AlertDialogTitle>
   315→            <AlertDialogDescription className="text-slate-400">
   316→              This will permanently delete <span className="text-rose-400 font-medium">{selectedUser?.email}</span> and all their resumes. This action cannot be undone.
   317→            </AlertDialogDescription>
   318→          </AlertDialogHeader>
   319→          <AlertDialogFooter>
   320→            <AlertDialogCancel className="bg-slate-800 border-slate-700 text-slate-300">Cancel</AlertDialogCancel>
   321→            <AlertDialogAction onClick={handleDelete} className="bg-rose-600 hover:bg-rose-700">Delete permanently</AlertDialogAction>
   322→          </AlertDialogFooter>
   323→        </AlertDialogContent>
   324→      </AlertDialog>
   325→    </div>
   326→  );
   327→}

   329→// User detail dialog — shows user info + their resumes (with view/edit)
   330→function UserDetailDialog({ user, token, open, onOpenChange, onEdit, onDelete }: {
   331→  user: AdminUser; token: string; open: boolean; onOpenChange: (v: boolean) => void; onEdit: () => void; onDelete: () => void;
   332→}) {
   333→  const [resumes, setResumes] = useState<{ id: string; title: string; template: string; createdAt: string; updatedAt: string; contactLocked: boolean }[]>([]);
   334→  const [loading, setLoading] = useState(true);
   335→
   336→  useEffect(() => {
   337→    if (!open) return;
   338→    let cancelled = false;
   339→    fetch(`/api/admin/users/${user.id}/resumes`, { headers: authHeaders(token) })
   340→      .then((r) => r.json())
   341→      .then((j) => { if (!cancelled) setResumes(j.resumes || []); })
   342→      .catch(() => { if (!cancelled) toast.error("Failed to load resumes"); })
   343→      .finally(() => { if (!cancelled) setLoading(false); });
   344→    return () => { cancelled = true; };
   345→  }, [open, user.id, token]);
   346→
   347→  const copyCreds = () => {
   348→    navigator.clipboard.writeText(`${user.email}`);
   349→    toast.success("Email copied");
   350→  };
   351→
   352→  return (
   353→    <Dialog open={open} onOpenChange={onOpenChange}>
   354→      <DialogContent className="max-w-3xl max-h-[85vh] overflow-hidden bg-slate-900 border-slate-700 flex flex-col">
   355→        <DialogHeader>
   356→          <DialogTitle className="text-white flex items-center gap-2">
   357→            <div className="w-8 h-8 rounded-full bg-slate-800 flex items-center justify-center text-sm">{user.email[0]?.toUpperCase()}</div>
   358→            User Details
   359→          </DialogTitle>
   360→          <DialogDescription className="text-slate-400">Full account info and resume management</DialogDescription>
   361→        </DialogHeader>
   362→        <div className="overflow-y-auto pr-1 space-y-4">
   363→          {/* Info grid */}
   364→          <div className="grid grid-cols-2 gap-3">
   365→            <InfoRow label="Email" value={user.email} onCopy={copyCreds} />
   366→            <InfoRow label="Name" value={user.name || "—"} />
   367→            <InfoRow label="Plan" value={<Badge className={`border-0 ${planBadge(user.plan)}`}>{planLabel(user.plan)}</Badge>} />
   368→            <InfoRow label="Role" value={user.role} />
   369→            <InfoRow label="Student ID" value={user.studentId || "—"} />
   370→            <InfoRow label="Organization" value={user.organization?.name || "—"} />
   371→            <InfoRow label="Plan Expires" value={user.planExpiresAt ? new Date(user.planExpiresAt).toLocaleString() : "No expiry"} />
   372→            <InfoRow label="Joined" value={new Date(user.createdAt).toLocaleString()} />
   373→          </div>
   374→
   375→          {/* Resumes */}
   376→          <div>
   377→            <p className="text-xs font-semibold text-slate-300 mb-2 flex items-center gap-1.5"><FileText className="w-3.5 h-3.5" /> Resumes ({resumes.length})</p>
   378→            {loading ? (
   379→              <div className="flex justify-center py-4"><Loader2 className="w-4 h-4 animate-spin text-teal-400" /></div>
   380→            ) : resumes.length === 0 ? (
   381→              <p className="text-center text-xs text-slate-400 py-6 bg-slate-800/30 rounded-lg">No resumes created yet.</p>
   382→            ) : (
   383→              <div className="space-y-2 max-h-60 overflow-y-auto">
   384→                {resumes.map((r) => (
   385→                  <div key={r.id} className="flex items-center gap-3 p-2.5 rounded-lg bg-slate-800/50">
   386→                    <FileText className="w-4 h-4 text-teal-400 shrink-0" />
   387→                    <div className="flex-1 min-w-0">
   388→                      <p className="text-xs font-medium truncate">{r.title}</p>
   389→                      <p className="text-[10px] text-slate-400">{r.template} · Updated {new Date(r.updatedAt).toLocaleDateString()}{r.contactLocked && " · 🔒 locked"}</p>
   390→                    </div>
   391→                    <Button size="sm" variant="ghost" className="h-7 text-[10px] gap-1 text-teal-400" onClick={() => window.open(`/r/${r.slug}`, "_blank")}>
   392→                      <ExternalLink className="w-3 h-3" /> Open
   393→                    </Button>
   394→                  </div>
   395→                ))}
   396→              </div>
   397→            )}
   398→          </div>
   399→        </div>
   400→        <DialogFooter className="gap-2">
   401→          <Button variant="outline" onClick={onDelete} className="gap-1.5 bg-rose-900/30 border-rose-800 text-rose-300 hover:bg-rose-900/50">

  1379→function ManageStudentsDialog({ org, token, open, onOpenChange, onCreated }: {

  1452→        <div className="flex items-center gap-2 mb-3">
  1453→          <button onClick={() => setMode("single")} className={`px-3 py-1.5 rounded text-xs ${mode === "single" ? "bg-violet-700 text-white" : "bg-slate-800 text-slate-400"}`}>Single</button>
